cmake_minimum_required(VERSION 3.19)
project(PcOriginalEffectFeasibility LANGUAGES C)

if(NOT DEFINED PC_UNICORN_SOURCE OR NOT EXISTS "${PC_UNICORN_SOURCE}/include/unicorn/unicorn.h")
    message(FATAL_ERROR "Provide the pinned local Unicorn2.1.4 source checkout")
endif()
set(UNICORN_ARCH "x86" CACHE STRING "Original effect evaluator architecture" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "Probe has independent source-state checks" FORCE)
set(UNICORN_BUILD_SAMPLES OFF CACHE BOOL "Do not ship unrelated emulation samples" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "No global tool/library installation" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Keep the emulation dependency separate" FORCE)
# Unicorn2.1.4's configure subprocess does not propagate a nonzero exit code.
# Reject missing discovery tools before it can generate an empty config header.
if(DEFINED ENV{PKG_CONFIG})
    set(PC_PKG_CONFIG "$ENV{PKG_CONFIG}")
else()
    find_program(PC_PKG_CONFIG NAMES pkg-config pkgconf REQUIRED)
endif()
execute_process(COMMAND "${PC_PKG_CONFIG}" --version
    RESULT_VARIABLE PC_PKG_CONFIG_RESULT OUTPUT_QUIET ERROR_QUIET)
if(NOT PC_PKG_CONFIG_RESULT EQUAL 0)
    message(FATAL_ERROR "A working explicit pkg-config implementation is required")
endif()
add_subdirectory("${PC_UNICORN_SOURCE}" "${CMAKE_BINARY_DIR}/unicorn")
file(READ "${CMAKE_BINARY_DIR}/config-host.h" PC_UNICORN_CONFIG)
if(NOT PC_UNICORN_CONFIG MATCHES "#define CONFIG_POSIX 1")
    message(FATAL_ERROR "Unicorn configure failed: config-host.h has no POSIX configuration; inspect config.log")
endif()
foreach(PC_PROBE pc_effect_vm_probe pc_effect_scene_probe)
    add_executable(${PC_PROBE} ${PC_PROBE}.c)
    target_include_directories(${PC_PROBE} PRIVATE "${PC_UNICORN_SOURCE}/include")
    target_compile_features(${PC_PROBE} PRIVATE c_std_11)
    target_compile_options(${PC_PROBE} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${PC_PROBE} PRIVATE unicorn m)
endforeach()
add_executable(pc_effect_worker pc_effect_scene_probe.c)
set_target_properties(pc_effect_worker PROPERTIES OUTPUT_NAME libpc_effect_worker SUFFIX .so)
target_include_directories(pc_effect_worker PRIVATE "${PC_UNICORN_SOURCE}/include")
target_compile_features(pc_effect_worker PRIVATE c_std_11)
target_compile_options(pc_effect_worker PRIVATE -Wall -Wextra -Werror)
target_link_libraries(pc_effect_worker PRIVATE unicorn m)
